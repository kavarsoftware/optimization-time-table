package edu.timetable;
import java.util.*;
import java.util.function.*;
class SearchEngine implements SearchMethod {
    static final org.slf4j.Logger LOG=org.slf4j.LoggerFactory.getLogger(SearchEngine.class);
    record Settings(int seed,double seconds,int population,int generations,double mutation,long samples){
        Settings{ProblemData.require(seconds>0&&Double.isFinite(seconds)&&population>=2&&generations>0&&mutation>=0&&mutation<=1&&samples>0,"Invalid search settings");}
    }
    interface ConstraintSolver {int[] solve(SearchEngine run,int[] child,boolean optimize,double seconds);}
    final ProblemData m;final Settings settings;final Random rng;final BooleanSupplier cancelled;final long start;
    long evaluations,mutations,nodes,attempts;
    boolean feasibleGenerator=true,warmStart=true,reportDiagnostics=true;
    int workers=Math.min(8,Runtime.getRuntime().availableProcessors());
    int[] hint;volatile int[] best;volatile int score=Integer.MAX_VALUE;String status="Budget ended";boolean proved;
    final List<double[]> trace=new ArrayList<>();
    SearchEngine(ProblemData model,Settings options,BooleanSupplier cancel){m=model;settings=options;rng=new Random(options.seed);cancelled=cancel;start=System.nanoTime();}
    double elapsed(){return (System.nanoTime()-start)/1e9;}
    double left(){return settings.seconds-elapsed();}
    boolean stop(){return left()<=0||cancelled.getAsBoolean();}
    int inspect(int[] genes){
        evaluations++;var result=new DefaultEvaluator().evaluate(m,new Candidate(genes));
        if(!result.feasible())return Integer.MAX_VALUE;
        if(result.objective()<score){best=genes.clone();score=result.objective();trace.add(new double[]{evaluations,score,elapsed()});}
        return result.objective();
    }
    int[] sample(){int[] genes=new int[m.events.size()];for(int i=0;i<genes.length;i++)genes[i]=rng.nextInt(m.choices.get(i).size());return genes;}
    public void run(String method,ConstraintSolver cp){
        LOG.debug("Starting {} seed={} budget={}",method,settings.seed,settings.seconds);
        for(int i=0;i<m.events.size();i++)if(m.choices.get(i).isEmpty()){
            var e=m.events.get(i);status="INFEASIBLE: "+m.diagnostics.get(i);return;}
        if(reportDiagnostics)diagnostics();
        switch(method){
            case "Random"->{while(!stop()&&attempts<settings.samples){attempts++;int[] candidate=feasibleGenerator?greedySeed():sample();if(candidate!=null)inspect(candidate);}}
            case "Exhaustive"->{int[] genes=new int[m.events.size()];Arrays.fill(genes,-1);Integer[] order=new Integer[genes.length];for(int i=0;i<order.length;i++)order[i]=i;Arrays.sort(order,Comparator.comparingInt(i->m.choices.get(i).size()));proved=enumerate(genes,order,0);}
            case "CP"->{if(warmStart){double until=elapsed()+Math.max(0,left()*.10);while(!stop()&&elapsed()<until){int[] seed=greedySeed();if(seed!=null)inspect(seed);}hint=best==null?null:best.clone();}int[] result=cp.solve(this,null,true,left());if(result!=null)inspect(result);}
            case "GA"->evolve(cp);
            case "Hybrid"->lns(cp);
            default->throw new IllegalArgumentException(method);
        }
        if(cancelled.getAsBoolean())status="Cancelled";
        else if(best!=null)status=proved||(!method.equals("Exhaustive")&&score==0)?"OPTIMAL":"FEASIBLE";
        else if(proved)status="INFEASIBLE";
        else status="UNKNOWN";
        LOG.debug("Finished {} status={} score={} seconds={}",method,status,score,elapsed());
    }
    void lns(ConstraintSolver cp){
        int[] seed=cp.solve(this,null,false,Math.min(left(),Math.max(.2,settings.seconds/3)));
        if(seed==null)return;inspect(seed);
        for(int i=0;i<settings.generations&&!stop()&&score>0;i++){
            int[] improved=cp.solve(this,best.clone(),true,Math.min(left(),Math.max(.2,settings.seconds/10)));
            if(improved!=null)inspect(improved);
        }
    }
    double searchSpaceLog10(){return m.choices.stream().mapToDouble(c->c.isEmpty()?Double.NEGATIVE_INFINITY:Math.log10(c.size())).sum();}
    void diagnostics(){
        var random=new Random(settings.seed());int feasible=0;
        for(int sample=0;sample<2000;sample++){int[] genes=new int[m.events.size()];for(int i=0;i<genes.length;i++)genes[i]=random.nextInt(m.choices.get(i).size());if(Evaluator.evaluate(m,genes).valid())feasible++;}
        System.out.printf(Locale.ROOT,"Search space: 10^%.3f; uniform feasible: %d/2000 (%.4f%%)%n",searchSpaceLog10(),feasible,feasible/20.0);
    }
    boolean enumerate(int[] genes,Integer[] order,int depth){
        if(stop())return false;
        if(depth==order.length){inspect(genes);return true;}
        int event=order[depth];
        for(int k=0;k<m.choices.get(event).size();k++){
            if(stop())return false;nodes++;boolean fits=true;
            for(int j=0;j<depth;j++){int previous=order[j];if(m.conflict(event,m.choices.get(event).get(k),previous,m.choices.get(previous).get(genes[previous]))!=null){fits=false;break;}}
            if(!fits)continue;genes[event]=k;
            if(!enumerate(genes,order,depth+1)){genes[event]=-1;return false;}
            genes[event]=-1;
        }
        return true;
    }
    record Member(int[] genes,int cost){}
    Member tournament(List<Member> population){
        Member winner=population.get(rng.nextInt(population.size()));
        for(int i=1;i<3;i++){Member rival=population.get(rng.nextInt(population.size()));if(rival.cost<winner.cost)winner=rival;}
        return winner;
    }
    int[] crossover(Member a,Member b){int[] child=a.genes.clone();for(int i=0;i<child.length;i++)if(rng.nextBoolean())child[i]=b.genes[i];return child;}
    void mutate(int[] child){
        for(int i=0;i<child.length;i++)if(rng.nextDouble()<settings.mutation){
            int size=m.choices.get(i).size();if(size<2)continue;
            int replacement=rng.nextInt(size-1);if(replacement>=child[i])replacement++;
            child[i]=replacement;mutations++;
        }
    }
    int[] greedySeed(){
        int[] genes=new int[m.events.size()];Arrays.fill(genes,-1);
        var order=new ArrayList<Integer>();for(int i=0;i<genes.length;i++)order.add(i);
        Collections.shuffle(order,rng);order.sort(Comparator.comparingInt(i->m.choices.get(i).size()));
        for(int i:order){
            var options=new ArrayList<Integer>();for(int k=0;k<m.choices.get(i).size();k++)options.add(k);
            Collections.shuffle(options,rng);boolean placed=false;
            for(int k:options){
                if(stop())return null;boolean fits=true;
                for(int j=0;j<genes.length;j++)if(genes[j]>=0&&m.conflict(i,m.choices.get(i).get(k),j,m.choices.get(j).get(genes[j]))!=null){fits=false;break;}
                if(fits){genes[i]=k;placed=true;break;}
            }
            if(!placed)return null;
        }
        return Evaluator.evaluate(m,genes).valid()?genes:null;
    }
    void evolve(ConstraintSolver cp){
        var population=new ArrayList<Member>();
        // Construct diverse feasible seeds cheaply before spending a CP budget.
        for(int attempt=0;attempt<settings.population&&!stop();attempt++){
            int[] seed=greedySeed();if(seed!=null){int cost=inspect(seed);if(cost!=Integer.MAX_VALUE)population.add(new Member(seed,cost));}
        }
        if(population.isEmpty()&&!stop()){
            int[] seed=cp.solve(this,null,true,Math.min(left(),Math.max(.2,left()*.70)));
            if(seed!=null){int cost=inspect(seed);if(cost!=Integer.MAX_VALUE)population.add(new Member(seed,cost));}
        }
        if(score==0)return;
        if(population.isEmpty())return;
        while(population.size()<settings.population)population.add(population.get(rng.nextInt(population.size())));
        for(int g=0;g<settings.generations&&!stop()&&score>0;g++){
            var next=new ArrayList<Member>();next.add(new Member(best.clone(),score));
            while(next.size()<settings.population&&!stop()){
                Member parent=tournament(population);int[] child=crossover(parent,tournament(population));mutate(child);
                int cost=inspect(child);
                if(cost==Integer.MAX_VALUE){child=cp.solve(this,child,false,Math.min(.3,left()));if(child!=null)cost=inspect(child);}
                next.add(cost<Integer.MAX_VALUE?new Member(child,cost):parent);
            }
            population=next;
        }
    }
}

package edu.timetable;
import com.google.ortools.sat.*;
import java.util.*;
/** Interval CP model. Resource conflicts use NoOverlap, never placement-pair clauses. */
class ConstraintSearch implements SearchEngine.ConstraintSolver {
    static {NativeSolver.load();}
    static LinearExpr offset(IntVar start,int value){return LinearExpr.newBuilder().add(start).add(value).build();}
    static void resource(Map<String,List<IntervalVar>> map,String key,IntervalVar interval){map.computeIfAbsent(key,k->new ArrayList<>()).add(interval);}
    public int[] solve(SearchEngine run,int[] child,boolean optimize,double seconds){
        long deadline=System.nanoTime()+(long)(seconds*1e9);ProblemData m=run.m;CpModel cp=new CpModel();int n=m.events.size();
        BoolVar[][] take=new BoolVar[n][];IntVar[] start=new IntVar[n],day=new IntVar[n];
        Map<String,List<IntervalVar>> resources=new LinkedHashMap<>();
        boolean[] move=new boolean[n];
        if(child!=null){
            if(optimize){for(int i=0;i<n;i++)move[i]=run.rng.nextDouble()<.3;move[run.rng.nextInt(n)]=true;}
            else for(int i=0;i<n;i++)for(int j=0;j<i;j++)if(m.conflict(i,m.choices.get(i).get(child[i]),j,m.choices.get(j).get(child[j]))!=null)move[i]=move[j]=true;
            if(!optimize&&!Evaluator.evaluate(m,child).valid()&&!any(move))Arrays.fill(move,true);
        }
        for(int i=0;i<n;i++){
            if(run.stop()||System.nanoTime()>deadline)return null;
            var e=m.events.get(i);take[i]=new BoolVar[m.choices.get(i).size()];
            var starts=LinearExpr.newBuilder();var days=LinearExpr.newBuilder();
            Map<String,List<BoolVar>> rooms=new LinkedHashMap<>();
            for(int k=0;k<take[i].length;k++){
                var p=m.choices.get(i).get(k);var selected=cp.newBoolVar("e"+i+"p"+k);take[i][k]=selected;
                int d=m.days.indexOf(p.day());starts.addTerm(selected,d*2880+p.start());days.addTerm(selected,d);
                rooms.computeIfAbsent(p.room(),r->new ArrayList<>()).add(selected);
            }
            cp.addExactlyOne(take[i]);start[i]=cp.newIntVar(0,m.days.size()*2880,"start"+i);day[i]=cp.newIntVar(0,m.days.size()-1,"day"+i);
            cp.addEquality(start[i],starts);cp.addEquality(day[i],days);
            if(child!=null&&!move[i])cp.addEquality(take[i][child[i]],1);
            var teaching=cp.newFixedSizeIntervalVar(start[i],e.span(),"class"+i);
            resource(resources,"teacher:"+e.teacher(),teaching);
            var attending=cp.newFixedSizeIntervalVar(start[i],e.span()+e.rest(),"attendance"+i);
            for(String group:e.groups())resource(resources,"group:"+group,attending);
            for(String student:e.students())resource(resources,"student:"+student,attending);
            for(var room:rooms.entrySet()){
                var selected=cp.newBoolVar("room"+i+room.getKey());cp.addEquality(selected,LinearExpr.sum(room.getValue().toArray(BoolVar[]::new)));
                for(var range:e.roomIntervals(0))
                    resource(resources,"room:"+room.getKey(),cp.newOptionalFixedSizeIntervalVar(offset(start[i],range[0]),range[1]-range[0],selected,"roomInterval"+i));
            }
        }
        // Simple mode has no post-class rest: blocked periods can also be explicit fixed intervals.
        // Legacy rest policies remain enforced by their original domain/attendance semantics.
        if(m.events.stream().allMatch(e->e.pause()==0))for(var p:m.periods)if(!p[4].equals("Teaching")){
            int begin=m.days.indexOf(p[0])*2880+Data.minute(p[2]);
            var blocked=cp.newFixedSizeIntervalVar(cp.newConstant(begin),Data.minute(p[3])-Data.minute(p[2]),"blockedPeriod");
            resources.values().forEach(intervals->intervals.add(blocked));
        }
        for(var intervals:resources.values())if(intervals.size()>1)cp.addNoOverlap(intervals);
        addRules(cp,m,start,day);
        workload(cp,m,take);
        if(optimize)addObjective(cp,m,take);
        else if(child==null)for(int i=0;i<n;i++)cp.addHint(take[i][run.rng.nextInt(take[i].length)],1);
        if(child==null&&run.hint!=null)for(int i=0;i<n;i++)for(int k=0;k<take[i].length;k++)cp.addHint(take[i][k],k==run.hint[i]?1:0);
        double remaining=Math.min(run.left(),(deadline-System.nanoTime())/1e9);if(remaining<=0||run.stop())return null;
        CpSolver solver=new CpSolver();solver.getParameters().setMaxTimeInSeconds(remaining).setNumSearchWorkers(run.workers).setRandomSeed(run.settings.seed());
        final RuntimeException[] mismatch={null};
        var callback=new CpSolverSolutionCallback(){
            public void onSolutionCallback(){
                int[] candidate=new int[n];for(int i=0;i<n;i++)for(int k=0;k<take[i].length;k++)if(booleanValue(take[i][k]))candidate[i]=k;
                var evaluation=Evaluator.evaluate(m,candidate);
                if(!evaluation.valid()||optimize&&!objectiveAgrees(objectiveValue(),evaluation.minutes(),false)){
                    mismatch[0]=new IllegalStateException("CP/evaluator disagreement: "+evaluation.violations()+" CP="+objectiveValue()+" evaluator="+evaluation.minutes());stopSearch();return;
                }
                run.inspect(candidate);if(run.cancelled.getAsBoolean())stopSearch();
            }
        };
        Thread watcher=Thread.ofVirtual().start(()->{try{while(!Thread.currentThread().isInterrupted()){if(run.cancelled.getAsBoolean())solver.stopSearch();Thread.sleep(40);}}catch(InterruptedException ignored){Thread.currentThread().interrupt();}});
        CpSolverStatus result;
        try{result=solver.solve(cp,callback);}finally{watcher.interrupt();}
        if(mismatch[0]!=null)throw mismatch[0];
        if(result==CpSolverStatus.INFEASIBLE&&child==null)run.proved=true;
        if(result!=CpSolverStatus.FEASIBLE&&result!=CpSolverStatus.OPTIMAL)return null;
        if(optimize&&child==null&&result==CpSolverStatus.OPTIMAL)run.proved=true;
        int[] answer=new int[n];for(int i=0;i<n;i++)for(int k=0;k<take[i].length;k++)if(solver.booleanValue(take[i][k]))answer[i]=k;
        // CP-SAT may report a loose auxiliary objective for intermediate incumbents.
        // The evaluator gives the exact score of their selected timetable. At proven
        // optimality the auxiliary objective must be tight; never conceal a mismatch.
        var evaluation=Evaluator.evaluate(m,answer);
        if(!evaluation.valid()||optimize&&!objectiveAgrees(solver.objectiveValue(),evaluation.minutes(),result==CpSolverStatus.OPTIMAL))
            throw new IllegalStateException("Final CP/evaluator disagreement: CP="+solver.objectiveValue()+" evaluator="+evaluation.minutes());
        return answer;
    }
    static boolean objectiveAgrees(double cpScore,int exactScore,boolean optimal){
        long reported=Math.round(cpScore);
        return optimal?reported==exactScore:reported>=exactScore;
    }
    static boolean any(boolean[] b){for(boolean v:b)if(v)return true;return false;}
    void addRules(CpModel cp,ProblemData m,IntVar[] start,IntVar[] day){
        for(var r:m.rules)for(int i=0;i<m.events.size();i++)for(int j=0;j<m.events.size();j++){
            if(i==j)continue;var a=m.events.get(i);var b=m.events.get(j);
            String second=r[1].equals("once daily")?r[2]:r[3];
            if(!a.code().equals(r[2])||!b.code().equals(second)||!r[4].isBlank()&&!(a.groups().contains(r[4])&&b.groups().contains(r[4])))continue;
            switch(r[1]){
                case "different days","once daily"->cp.addDifferent(day[i],day[j]);
                case "no overlap"->cp.addNoOverlap(new IntervalVar[]{cp.newFixedSizeIntervalVar(start[i],a.span(),"ruleA"),cp.newFixedSizeIntervalVar(start[j],b.span(),"ruleB")});
                case "after"->cp.addGreaterOrEqual(start[j],offset(start[i],a.span()+a.rest()));
                case "consecutive"->{cp.addEquality(day[i],day[j]);cp.addEquality(start[j],offset(start[i],a.span()+a.rest()));}
            }
        }
    }
    LinearExprBuilder sumFor(ProblemData m,BoolVar[][] take,String teacher,String day){
        var sum=LinearExpr.newBuilder();for(int i=0;i<m.events.size();i++)if(m.events.get(i).teacher().equals(teacher))
            for(int k=0;k<take[i].length;k++)if(day==null||m.choices.get(i).get(k).day().equals(day))sum.addTerm(take[i][k],Objectives.teaching(m.events.get(i)));
        return sum;
    }
    void workload(CpModel cp,ProblemData m,BoolVar[][] take){
        for(var t:m.data.rows.get("Workloads")){
            if(!t[2].isBlank())cp.addLessOrEqual(sumFor(m,take,t[0],null),Data.positive(t[2]));
            if(!t[1].isBlank())for(String day:m.days)cp.addLessOrEqual(sumFor(m,take,t[0],day),Data.positive(t[1]));
        }
    }
    void addObjective(CpModel cp,ProblemData m,BoolVar[][] take){
        int[] w=Objectives.weights(m);var objective=LinearExpr.newBuilder();Set<String> groups=new TreeSet<>(),teachers=new TreeSet<>();
        m.events.forEach(e->{groups.addAll(m.audience(e));teachers.add(e.teacher());});
        for(String group:groups)scopes(cp,m,take,objective,group,false,w);
        if(w[1]>0)for(String teacher:teachers)scopes(cp,m,take,objective,teacher,true,w);
        var rooms=new TreeMap<String,Integer>();for(var room:m.data.rows.get("Rooms"))rooms.put(room[0],Data.positive(room[2]));
        var prefs=Objectives.preferences(m);
        Map<String,Map<String,List<BoolVar>>> subjectRooms=new TreeMap<>();
        for(int i=0;i<m.events.size();i++){var e=m.events.get(i);
            for(int k=0;k<take[i].length;k++){
                var p=m.choices.get(i).get(k);long penalty=w[3]*(Math.max(0,Data.minute(prefs[0])-p.start())+Math.max(0,p.start()+e.span()-Data.minute(prefs[1])));
                penalty+=(long)w[4]*Math.max(0,Math.min(p.start()+e.span(),Data.minute(prefs[3]))-Math.max(p.start(),Data.minute(prefs[2])));
                penalty+=(long)w[7]*(rooms.get(p.room())-e.seats());
                for(var t:m.data.rows.get("Workloads"))if(t[0].equals(e.teacher())&&!t[3].isBlank()&&!Data.names(t[3]).contains(p.day()))penalty+=w[8];
                objective.addTerm(take[i][k],penalty);
                if(w[6]>0)subjectRooms.computeIfAbsent(e.code(),c->new TreeMap<>()).computeIfAbsent(p.room(),r->new ArrayList<>()).add(take[i][k]);
            }
        }
        for(var roomChoices:subjectRooms.values()){
            objective.add(-w[6]);for(var choices:roomChoices.values()){
                var used=cp.newBoolVar("usedRoom");var sum=LinearExpr.sum(choices.toArray(BoolVar[]::new));
                cp.addGreaterOrEqual(sum,1).onlyEnforceIf(used);cp.addEquality(sum,0).onlyEnforceIf(used.not());objective.addTerm(used,w[6]);
            }
        }
        cp.minimize(objective);
    }
    void scopes(CpModel cp,ProblemData m,BoolVar[][] take,LinearExprBuilder objective,String id,boolean teacher,int[] w){
        int total=0;for(var e:m.events)if(teacher?e.teacher().equals(id):m.audience(e).contains(id))total+=Objectives.teaching(e);
        for(String day:m.days){
            var count=LinearExpr.newBuilder();var load=LinearExpr.newBuilder();var occupied=LinearExpr.newBuilder();
            int weight=teacher?w[1]:w[0];var firstChoices=new ArrayList<IntVar>();var lastChoices=new ArrayList<IntVar>();
            for(int i=0;i<m.events.size();i++){
                var e=m.events.get(i);if(!(teacher?e.teacher().equals(id):m.audience(e).contains(id)))continue;
                var onDay=LinearExpr.newBuilder();var begins=LinearExpr.newBuilder();var ends=LinearExpr.newBuilder();
                for(int k=0;k<take[i].length;k++){
                    var p=m.choices.get(i).get(k);if(!p.day().equals(day))continue;
                    int begin=m.activeMinute(day,p.start()),end=m.activeMinute(day,p.start()+e.span()+(teacher?0:e.rest()));
                    occupied.addTerm(take[i][k],end-begin);count.add(take[i][k]);load.addTerm(take[i][k],Objectives.teaching(e));
                    onDay.add(take[i][k]);begins.addTerm(take[i][k],begin);ends.addTerm(take[i][k],end);
                }
                if(weight>0){
                    var present=cp.newBoolVar("onDay");cp.addEquality(present,onDay);
                    var begin=cp.newIntVar(0,2880,"beginOrSentinel");cp.addEquality(begin,begins.add(2880).addTerm(present,-2880));firstChoices.add(begin);
                    var end=cp.newIntVar(0,2880,"endOrZero");cp.addEquality(end,ends);lastChoices.add(end);
                }
            }
            if(weight>0&&!firstChoices.isEmpty()){
                var rawFirst=cp.newIntVar(0,2880,"rawFirst");var first=cp.newIntVar(0,2880,"first");var last=cp.newIntVar(0,2880,"last");
                cp.addMinEquality(rawFirst,firstChoices);cp.addMaxEquality(last,lastChoices);
                var any=cp.newBoolVar("any");cp.addGreaterOrEqual(count,1).onlyEnforceIf(any);cp.addEquality(count,0).onlyEnforceIf(any.not());
                cp.addEquality(first,rawFirst).onlyEnforceIf(any);cp.addEquality(first,0).onlyEnforceIf(any.not());
                objective.addTerm(last,weight).addTerm(first,-weight).addTerm(occupied,-weight);
            }
            if(!teacher&&w[5]>0){var single=cp.newBoolVar("single");cp.addEquality(count,1).onlyEnforceIf(single);cp.addDifferent(count,1).onlyEnforceIf(single.not());objective.addTerm(single,w[5]);}
            if(!teacher&&w[2]>0){
                var difference=LinearExpr.newBuilder().addTerm(load,m.days.size()).add(-total);
                var abs=cp.newIntVar(0,(long)total*m.days.size(),"balance");cp.addAbsEquality(abs,difference);objective.addTerm(abs,w[2]);
            }
        }
    }
}

# UniPlan5 → UniPlan Simple

## Kept

- ProblemData, Candidate, Evaluation, Evaluator/DefaultEvaluator, Objectives, SearchEngine and interval-based ConstraintSearch.
- Independent feasibility/objective verification, seeded Random and Exhaustive baselines, CP-SAT, JSON persistence pattern and autosave.
- Week grid, filters, printable HTML, CSV and a single green palette.
- All eight existing JUnit tests, with new Simple-specific tests added. CLI experiments remain available.

## Removed from the product UI

- Semester dates/holidays/exam ranges and ICS.
- Workloads, preferred days, weights, raw tables and lock/rule editors.
- GA/Hybrid choices and parameter fields; experiment charts/tables.
- Multiple-semester portfolio switching, alternative timetables and manual drag/drop.
- Old Studio/Setup/RecordForm/TermEditor/ObjectiveEditor frontend classes and its old smoke harness. CalendarExport and legacy JSON/model helpers remain only because the retained tests/developer tools exercise them.

## Added

- Seven-screen SimpleApp with left stepper, typed labeled controls, inline errors, tooltips and editing dialogs.
- Fixed stages 1–4, editable group lists, per-group subjects with independent enrollment/session count/duration and optional instructor.
- Independent daily start/end/class length/breaks, copy action, live green/grey preview.
- SimpleProject adapter/document schema 1, explicit validation and fixed gap/balance weights.
- Friendly health report with linked fixes, CP default, 20-second limit, responsive progress and Cancel.
- 4-stage/8-group demo with 56 sessions, 10 classrooms and 4 labs.
- Subject IDs independent of names, unique internal placeholders for blank instructors, no false collision between all unassigned teachers.
- Screenshots, real test/launch logs, data dictionary, research/credits and a three-minute explanation.

## Bugs fixed

- JavaFX 21.0.2's duplicate Maven profile IDs generated six warnings: upgraded the existing JavaFX dependency to 21.0.9.
- JavaFX Maven plugin 0.0.8 drops OR-Tools native JARs with invalid automatic module names in non-modular projects. Its includePathExceptions setting does not fix that branch. Native resources now unpack from the same pinned OR-Tools version during process-resources, for the selected OS only; they are not put through module-name inference.
- OR-Tools 9.11's built-in loader assumes a jar resource URI; it cannot unpack a plain classes-directory URI. The small NativeSolver adapter loads the JNI library directly for file resources, and delegates to the original loader for jar resources. No search algorithm was rewritten.
- HTML used stale navy colors and called a weighted score “gap minutes.” It now derives colors from studio.css and labels the weighted objective correctly.
- Simple mode adds fixed blocked intervals for breaks/remainders to each existing resource NoOverlap constraint. Legacy policies with post-class rest retain their original domain semantics, so those tests continue to pass.
- Error handling now logs the actual cause of a failed background task, including native initialization errors, instead of showing only “null.”

## Intentional choices

Daily class length defines full teaching periods; subject duration can be shorter but cannot cross periods. Short remainders are unavailable. New subjects default to the first selected day's class length. Exact semantics are shown in the UI and README.

The new JSON schema does not silently import advanced UniPlan5 settings. The old reader remains in the code for legacy tests/tools. Four-stage Simple inputs and legacy UniPlan5 inputs remain separate documents.

## Deliberately hidden/fixed values

The adapter sets zero per-subject rest (breaks come from daily inputs), 5-minute candidate-start steps inside each full class period, no calendar dates, no custom rules/locks/workloads, and fixed objective weights: student gaps = 1, daily balance = 1, all other terms = 0. The balance penalty is the existing engine's sum of |D × daily teaching minutes − weekly teaching minutes| per group/day. It encourages spreading lessons; it is not a hard attendance-day requirement.

A subject may be shorter than a day class period, but cannot cross its boundary. A 120-minute subject cannot use a day configured with 90-minute periods: Review explains which setting to change. Starts are selected every five minutes from the period's start, so a 09:10 period starts on its own grid, not on an assumed 09:00 grid.

Subjects have generated stable IDs in JSON, separate from their names. Group engine IDs combine stage and group name, e.g. `4 A`. Different groups are treated as disjoint student groups in Simple; cross-group individual student enrollment is not an input in this version. Use a separate subject row for each group. Stage A and B are not automatically combined into shared lectures.
